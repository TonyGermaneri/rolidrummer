cmake_minimum_required(VERSION 3.22)
if(APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET 11.0 CACHE STRING "Minimum macOS")
endif()
project(RoliDrummer VERSION 0.3.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(RD_BUILD_PLUGIN "Build AU/VST3/standalone" ON)
set(SSB_BUILD_PLUGIN ${RD_BUILD_PLUGIN})
include(CTest)
if(SSB_BUILD_PLUGIN)
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.15 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(JUCE)
 add_subdirectory(engine)
 if(APPLE)
  # Apple's universal system library is present in the SDK; Homebrew supplies
  # public headers omitted by recent SDKs. Do not link a single-arch brew dylib.
  find_path(RD_ARCHIVE_INCLUDE archive.h HINTS /usr/local/opt/libarchive/include /opt/homebrew/opt/libarchive/include REQUIRED)
  set(RD_ARCHIVE_LIB archive)
  set(RD_LIBRARY_SQLITE sqlite3)
 else()
  find_package(LibArchive REQUIRED)
  find_package(SQLite3 REQUIRED)
  set(RD_ARCHIVE_INCLUDE ${LibArchive_INCLUDE_DIRS})
  set(RD_ARCHIVE_LIB LibArchive::LibArchive)
  set(RD_LIBRARY_SQLITE SQLite::SQLite3)
 endif()
 juce_add_binary_data(DrummerMapsData HEADER_NAME DrumMapsData.h NAMESPACE drummermaps SOURCES ../src/data/drumMaps.json)
 add_library(drummer_library STATIC drummer/src/MidiImport.cpp drummer/src/PatternAnalysis.cpp drummer/src/PatternLibrary.cpp drummer/src/TextEmbedding.cpp drummer/src/VectorSearch.cpp drummer/src/NeuralIndex.cpp drummer/src/PatternFamilies.cpp drummer/src/PatchResolver.cpp drummer/src/PatternVariation.cpp)
 if(APPLE)
  enable_language(OBJCXX)
  set_source_files_properties(drummer/src/VectorSearch.cpp PROPERTIES LANGUAGE OBJCXX)
  target_link_libraries(drummer_library PRIVATE "-framework Metal")
 endif()
 target_include_directories(drummer_library PUBLIC drummer/include PRIVATE ${RD_ARCHIVE_INCLUDE})
 target_compile_definitions(drummer_library PUBLIC JUCE_USE_CURL=0)
 set(RD_NEURAL_ASSET_DIR "${CMAKE_CURRENT_SOURCE_DIR}/neural-assets" CACHE PATH "Pinned neural model and runtime assets")
 option(RD_REQUIRE_NEURAL_ASSETS "Fail configuration when neural assets are missing" ON)
 if(RD_REQUIRE_NEURAL_ASSETS AND (NOT EXISTS "${RD_NEURAL_ASSET_DIR}/manifest.json" OR NOT EXISTS "${RD_NEURAL_ASSET_DIR}/minilm/model.onnx" OR NOT EXISTS "${RD_NEURAL_ASSET_DIR}/onnxruntime/lib/libonnxruntime.1.23.1.dylib"))
  message(FATAL_ERROR "Pinned neural model/runtime assets are missing. Run python3 native/tools/fetch-neural-assets.py")
 endif()
 if(EXISTS "${RD_NEURAL_ASSET_DIR}/onnxruntime/include/onnxruntime_c_api.h")
  target_include_directories(drummer_library PRIVATE "${RD_NEURAL_ASSET_DIR}/onnxruntime/include")
  target_compile_definitions(drummer_library PRIVATE RD_HAVE_ONNX=1 RD_NEURAL_SOURCE_DIR="${RD_NEURAL_ASSET_DIR}")
 elseif(RD_REQUIRE_NEURAL_ASSETS)
  message(FATAL_ERROR "Run python3 native/tools/fetch-neural-assets.py before configuring the neural instrument")
 else()
  target_compile_definitions(drummer_library PRIVATE RD_HAVE_ONNX=0)
 endif()
 target_link_libraries(drummer_library PRIVATE DrummerMapsData ${RD_LIBRARY_SQLITE} ${RD_ARCHIVE_LIB} PUBLIC juce::juce_audio_basics juce::juce_cryptography)
 if(BUILD_TESTING)
  add_executable(drummer-library-tests drummer/tests/LibraryTest.cpp)
  target_link_libraries(drummer-library-tests PRIVATE drummer_library)
  add_test(NAME library COMMAND drummer-library-tests)
  add_executable(drummer-variation-tests drummer/tests/PatternVariationTest.cpp)
  target_link_libraries(drummer-variation-tests PRIVATE drummer_library)
  add_test(NAME variations COMMAND drummer-variation-tests)
  add_executable(drummer-vector-tests drummer/tests/VectorSearchTest.cpp)
  target_link_libraries(drummer-vector-tests PRIVATE drummer_library)
  add_test(NAME vectors COMMAND drummer-vector-tests)
  if(EXISTS "${RD_NEURAL_ASSET_DIR}/manifest.json")
   add_executable(drummer-embedding-tests drummer/tests/EmbeddingTest.cpp)
   target_link_libraries(drummer-embedding-tests PRIVATE drummer_library)
   add_test(NAME embedding COMMAND drummer-embedding-tests "${RD_NEURAL_ASSET_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/drummer/tests/data/tokenizer-reference.json")
   add_executable(drummer-neural-tests drummer/tests/NeuralIndexTest.cpp)
   target_link_libraries(drummer-neural-tests PRIVATE drummer_library)
   add_test(NAME neural COMMAND drummer-neural-tests "${RD_NEURAL_ASSET_DIR}")
   add_executable(drummer-family-tests drummer/tests/PatternFamiliesTest.cpp)
   target_link_libraries(drummer-family-tests PRIVATE drummer_library)
   add_test(NAME families COMMAND drummer-family-tests "${RD_NEURAL_ASSET_DIR}")
  endif()
 endif()
 add_executable(drummer-library tools/library.cpp)
 target_link_libraries(drummer-library PRIVATE drummer_library)
 add_executable(drummer-embedding tools/embedding.cpp)
 target_link_libraries(drummer-embedding PRIVATE drummer_library)
 if(EXISTS "${RD_NEURAL_ASSET_DIR}/onnxruntime/include/onnxruntime_float16.h")
  add_executable(drummer-vector-storage EXCLUDE_FROM_ALL tools/vector-storage.cpp)
  target_include_directories(drummer-vector-storage PRIVATE "${RD_NEURAL_ASSET_DIR}/onnxruntime/include")
  target_link_libraries(drummer-vector-storage PRIVATE drummer_library)
 endif()
endif()
add_subdirectory(core)
add_library(drummer_core STATIC drummer/src/Scheduler.cpp drummer/src/RhythmSpace.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
 set_property(SOURCE drummer/src/RhythmSpace.cpp drummer/src/PatternFamilies.cpp APPEND PROPERTY COMPILE_OPTIONS -ffp-contract=off)
endif()
target_include_directories(drummer_core PUBLIC drummer/include)
if(TARGET drummer_library)
 target_link_libraries(drummer_library PUBLIC drummer_core)
endif()
if(BUILD_TESTING)
 add_subdirectory(tests)
 add_executable(drummer-tests drummer/tests/SchedulerTest.cpp)
 target_link_libraries(drummer-tests PRIVATE drummer_core)
 add_test(NAME scheduler COMMAND drummer-tests)
 add_executable(drummer-intent-tests drummer/tests/IntentStressTest.cpp)
 target_link_libraries(drummer-intent-tests PRIVATE drummer_core)
 add_test(NAME intent COMMAND drummer-intent-tests)
 add_executable(drummer-rhythm-tests drummer/tests/RhythmSpaceTest.cpp)
 target_link_libraries(drummer-rhythm-tests PRIVATE drummer_core)
 add_test(NAME rhythm COMMAND drummer-rhythm-tests)
endif()
if(RD_BUILD_PLUGIN AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/plugin/CMakeLists.txt")
 set(SSB_WEB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
 include(cmake/WebAssets.cmake)
 add_subdirectory(plugin)
endif()
